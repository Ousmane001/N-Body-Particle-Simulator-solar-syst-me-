cmake_minimum_required(VERSION 3.18)
project(nbody_gpu VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(NBODY_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)

add_library(nbody_cpu_reference
    src/common/state.cpp
    src/common/diagnostics.cpp
    src/common/initial_conditions.cpp
    src/common/live.cpp
    src/cpu/forces.cpp
    src/cpu/simulation.cpp)
target_include_directories(nbody_cpu_reference PUBLIC include)
target_compile_features(nbody_cpu_reference PUBLIC cxx_std_17)
set_target_properties(nbody_cpu_reference PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(nbody_cpu_reference PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(NBODY_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "NBODY_SANITIZERS requires Clang or GCC")
    endif()
    target_compile_options(nbody_cpu_reference PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(nbody_cpu_reference PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nbody_cpu src/cpu/main.cpp)
target_link_libraries(nbody_cpu PRIVATE nbody_cpu_reference)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(nbody_cpu PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(nbody_tests tests/test_cpu.cpp)
    target_link_libraries(nbody_tests PRIVATE nbody_cpu_reference)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(nbody_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    foreach(case IN ITEMS forces softening validation free_motion circular_orbit second_order momentum energy reversibility initial_conditions)
        add_test(NAME cpu_${case} COMMAND nbody_tests ${case})
    endforeach()
    add_test(NAME cli_help COMMAND nbody_cpu --help)
    add_test(NAME cli_binary COMMAND nbody_cpu --scenario binary --steps 50)
    add_test(NAME cli_headless COMMAND nbody_cpu --scenario cluster --count 32 --steps 10)
    foreach(case IN ITEMS zero_dt nan_dt negative_count zero_steps unsupported_solver bad_scenario)
        if(case STREQUAL "zero_dt")
            set(args --dt 0)
        elseif(case STREQUAL "nan_dt")
            set(args --dt nan)
        elseif(case STREQUAL "negative_count")
            set(args --count -1)
        elseif(case STREQUAL "zero_steps")
            set(args --steps 0)
        elseif(case STREQUAL "unsupported_solver")
            set(args --solver barnes-hut)
        else()
            set(args --scenario unknown)
        endif()
        add_test(NAME cli_reject_${case} COMMAND nbody_cpu ${args})
        set_tests_properties(cli_reject_${case} PROPERTIES WILL_FAIL TRUE)
    endforeach()
endif()
